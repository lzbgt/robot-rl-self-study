% Keep the generated cover art separate from the typography: the image carries
% the visual theme, while TeX keeps the title, subtitle, and author exact and
% sharp at every zoom level. Defining \subtitle here makes Pandoc's later
% \providecommand store it independently instead of appending it to \@title.
\definecolor{CoverIvory}{HTML}{F4EFE3}
\definecolor{CoverAmber}{HTML}{C9994B}
\makeatletter
\providecommand{\@subtitle}{}
\providecommand{\subtitle}[1]{\gdef\@subtitle{#1}}
\renewcommand{\maketitle}{%
  \begin{titlepage}
    \thispagestyle{empty}%
    \AddToShipoutPictureBG*{%
      \AtPageLowerLeft{%
        \includegraphics[width=\paperwidth,height=\paperheight]{pdf/cover-art.png}%
      }%
    }%
    \centering
    \vspace*{5mm}%
    {\color{CoverIvory}\rmfamily\bfseries\fontsize{30}{35}\selectfont
      \@title\par}%
    \vspace{5mm}%
    {\color{CoverIvory}\sffamily\mdseries\fontsize{12.5}{16}\selectfont
      \@subtitle\par}%
    \vspace{31mm}%
    {\color{CoverAmber}\rule{58mm}{0.9pt}\par}%
    \vfill
    {\color{CoverIvory}\sffamily\bfseries\fontsize{15}{19}\selectfont
      \@author\par}%
    \vspace*{-2mm}%
  \end{titlepage}%
}
\makeatother

% Heading hierarchy is a navigation aid: chapters and conceptual subheads use
% navy, while section-level progress through a chapter uses teal. Size, weight,
% and rules duplicate those signals for grayscale readers.
\titleformat{\chapter}[display]
  {\normalfont\sffamily\bfseries\color{ConceptNavy}}
  {}{0pt}{\Huge}
  [\vspace{0.65ex}\color{ProcessTeal}\titlerule]
\titlespacing*{\chapter}{0pt}{-1.4em}{2.0em}
\titleformat{\section}
  {\normalfont\sffamily\bfseries\Large\color{ProcessTeal}}
  {}{0pt}{}
\titlespacing*{\section}{0pt}{1.7em}{0.55em}
\titleformat{\subsection}
  {\normalfont\sffamily\bfseries\large\color{ConceptNavy}}
  {}{0pt}{}
\titlespacing*{\subsection}{0pt}{1.35em}{0.4em}
\titleformat{\subsubsection}
  {\normalfont\sffamily\bfseries\normalsize\color{ConceptNavy}}
  {}{0pt}{}
\titlespacing*{\subsubsection}{0pt}{1.15em}{0.3em}

% Pandoc emits wide Markdown tables as longtable. Smaller cell padding and
% ragged-right paragraph columns prevent short technical tokens from producing
% large gaps or spilling into an adjacent column.
\AtBeginEnvironment{longtable}{%
  \small\sffamily
  \setlength{\tabcolsep}{3pt}%
  \renewcommand{\arraystretch}{1.14}%
  \arrayrulecolor{TableRule}%
}
\setlength{\LTleft}{0pt}
\setlength{\LTright}{0pt}

% GitHub hides reference answers in <details> elements. The PDF expands them
% and inserts this labeled banner. Only the banner is boxed, so long answers,
% tables, code, and equations remain free to break across pages.
\newcommand{\solutionheading}[1]{%
  \par\addvspace{1.35em}%
  \Needspace{6\baselineskip}%
  \begingroup
  \setlength{\fboxsep}{6pt}%
  \noindent\colorbox{AnswerTint}{%
    \parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
      \sffamily
      {\footnotesize\bfseries\color{AnswerAmber}%
        REFERENCE ANSWER · CHECK AFTER ATTEMPTING\par}%
      \vspace{0.25em}%
      {\normalsize\bfseries\color{ConceptNavy}#1\par}%
    }%
  }%
  \endgroup
  \par\nobreak\smallskip
}
